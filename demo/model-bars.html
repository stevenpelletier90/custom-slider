Model bar library

All 90 model bars across the 129 OEM demo sites are one design — arrows, swipe, one card per step, a sliver of the next card peeking, and never autoplay (the only autoplaying strips anywhere are one group site’s make and location rails — not model bars — shown last on this page). Brands differ in two ways only: the ladder (how many cards show at each width) and the look (what the cards wear — plain white, a gray or black band, a colour tile behind each car, photo cards). Every strip below runs live on a different brand’s ladder wearing that brand’s look; resize the window and watch the counts change.

Where one ladder ships with two looks in the estate, both are shown — same breakpoints, different clothes, same engine. Each strip carries everything its build needs. Tab script — once per page

<script>
  // Tab switching. Each pane holds its own slider; they look after themselves.
  addEventListener('DOMContentLoaded', function () {
    document.querySelectorAll('[data-tabs]').forEach(function (wrap) {
      var tabs = Array.prototype.slice.call(wrap.querySelectorAll('[role="tab"]'));
      var panes = tabs.map(function (t) { return document.getElementById(t.getAttribute('aria-controls')); });
      function select(i) {
        tabs.forEach(function (t, j) {
          t.setAttribute('aria-selected', String(i === j));
          t.tabIndex = i === j ? 0 : -1;
          panes[j].hidden = i !== j;
        });
        wrap.dataset.tab = i;
      }
      tabs.forEach(function (t, i) { t.addEventListener('click', function () { select(i); }); });
      wrap.querySelector('[role="tablist"]').addEventListener('keydown', function (e) {
        var i = tabs.indexOf(e.target); if (i === -1) return;
        var n = e.key === 'ArrowRight' ? (i + 1) % tabs.length : e.key === 'ArrowLeft' ? (i - 1 + tabs.length) % tabs.length : null;
        if (n === null) return;
        e.preventDefault(); tabs[n].focus(); select(n);
      });
      select(0);
    });
  });
</script>
